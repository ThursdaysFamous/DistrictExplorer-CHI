  /* =========================================================================
   * SIBLING-METRO HANDOFF — every instance knows where its siblings live
   * (METRO_EXPLORERS[].bbox). When the map centre, a searched address or a
   * geolocation fix lands inside a sibling's area, the reader goes STRAIGHT
   * there, carrying the zoom they were at (and the point, when there is one)
   * so the map they arrive on is framed like the one they left.
   *
   * This was an easter egg until 2026-08-25: a card, a five-second countdown
   * and a "Stay" button. Retired at the operator's direction — it asked a
   * question whose answer was always yes, and made every reader who crossed a
   * state line watch a timer to get where they were already going.
   * ======================================================================= */

  /* A metro without a (well-formed) bbox simply opts out of the handoff —
   * it still gets its footer link, so new METRO_EXPLORERS entries can land
   * before their bbox is decided. */
  function validMetroBbox(b) {
    return !!b && isFinite(b.minLat) && isFinite(b.maxLat) &&
      isFinite(b.minLng) && isFinite(b.maxLng) &&
      b.minLat < b.maxLat && b.minLng < b.maxLng;
  }

  function pointInMetroBbox(lat, lng, b) {
    return lat >= b.minLat && lat <= b.maxLat && lng >= b.minLng && lng <= b.maxLng;
  }

  function siblingMetroAt(lat, lng) {
    // greater-metro bboxes may overlap once enough metros exist (two nearby
    // cities) — the nearest bbox center wins, not list order. Squared-degree
    // distance is plenty for a tie-break between neighbors.
    var best = null, bestDist = Infinity;
    for (var i = 0; i < METRO_EXPLORERS.length; i++) {
      var m = METRO_EXPLORERS[i];
      if (m.id === THIS_METRO || !validMetroBbox(m.bbox)) continue;
      if (!pointInMetroBbox(lat, lng, m.bbox)) continue;
      var dLat = lat - (m.bbox.minLat + m.bbox.maxLat) / 2;
      var dLng = lng - (m.bbox.minLng + m.bbox.maxLng) / 2;
      var dist = dLat * dLat + dLng * dLng;
      if (dist < bestDist) { best = m; bestDist = dist; }
    }
    return best;
  }

  /* The sibling's URL, framed like the view being left: the point when there
   * is one (a click, a search hit, a geolocation fix) and the zoom always. A
   * pan-triggered handoff carries zoom alone — the reader selected nothing, so
   * arriving with a point selected for them would answer a question they never
   * asked. */
  function metroPortalUrl(metro, point, zoom) {
    var parts = [];
    if (point) parts.push("point=" + point.lat.toFixed(5) + "," + point.lng.toFixed(5));
    if (isFinite(zoom)) parts.push("zoom=" + zoom);
    return parts.length ? metro.url + "#" + parts.join("&") : metro.url;
  }

  // One handoff per page load. Belt and braces beside the coverage guard
  // below: whatever happens, a reader can never be bounced twice by one view.
  var metroHandedOff = false;

  /* Send the reader to a sibling instance. opts: {point, source} — point
   * deep-links via #point=, source labels the trackEvent. (title/note were the
   * retired card's copy; callers may still pass them and they are ignored.) */
  function offerMetroPortal(metro, opts) {
    opts = opts || {};
    if (!metro || metroHandedOff) return;
    var here = opts.point || null;
    // NEVER hand off a place this instance also answers for. Fleet bboxes
    // meet at shared borders — the Illinois line runs through both Chicago's
    // box and Wisconsin's — and a silent redirect on both sides of that strip
    // would bounce a reader back and forth with nothing on screen to stop it.
    // The card's "Stay" button used to be that stop; this is.
    if (here && pointInMetroBbox(here.lat, here.lng, METRO_BBOX)) return;
    metroHandedOff = true;
    trackEvent("metro-portal-go/" + metro.id, opts.source || "wander");
    window.location.href = metroPortalUrl(metro, here, map.getZoom());
  }

  map.on("moveend", function () {
    if (metroHandedOff) return;
    var c = map.getCenter();
    // The centre is a place this instance serves, so it is not a handoff — it
    // is a reader looking at their own map near its edge.
    if (pointInMetroBbox(c.lat, c.lng, METRO_BBOX)) return;
    var metro = siblingMetroAt(c.lat, c.lng);
    if (!metro) return;
    offerMetroPortal(metro, { source: "wander" });
  });
