# Pinned Python dependencies for this fork's tooling. Used as a constraints
# file: `pip install -c scripts/requirements.txt <pkg>` installs <pkg> at the
# version pinned here.
#
# SF has no HTML scrapers — its rosters are built from canonical machine-readable
# sources (OpenStates ca.csv, unitedstates/congress-legislators, DataSF) with the
# stdlib only, so the roster builders need nothing here. Two gates do:
#   scripts/validate_sources.py        (requests — source-freshness gate)
#   scripts/generate_metro_files.py    (jsonschema — worksheet validation)
# validate_index.py deliberately stays stdlib and does NOT import either.
requests==2.33.1
jsonschema==4.26.0

# scripts/build_sf_county_outline.py unions the eleven shipped supervisor
# districts into the city-and-county outline the Data gaps panel probes.
# Edge-cancellation (Wisconsin's approach for counties that tile on exact
# shared coordinates) is not safe here — the root repo's own requirements
# record it leaving a spurious 3-segment ring on precinct data — so this
# uses shapely's unary_union, at the version already pinned in scripts/
# and wi/scripts/.
shapely==2.1.2
